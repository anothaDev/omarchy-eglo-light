# Pinned, hash-verified runtime dependencies for the private venv that
# bin/lightctl builds under ~/.local/share/eglo-light/venv.
#
# The installer runs pip with --require-hashes --only-binary=:all: --no-deps
# --isolated, so only the exact wheels listed here can ever be installed: no
# unpinned upgrade, no sdist build step, no transitive package the maintainer
# has not reviewed, and no influence from the user's pip config or PIP_* env.
# That means every transitive dependency must be listed explicitly below.
#
# Covered platforms: Linux x86_64 and aarch64 (manylinux), CPython 3.13 and
# 3.14 (3.14 is the interpreter these pins were generated against; bleak and
# pycryptodome wheels are ABI-independent, dbus-fast is not). Older CPython
# (< 3.12) would additionally need typing-extensions and is not supported.
#
# To regenerate after a version bump:
#
#   for plat in manylinux2014_x86_64 manylinux2014_aarch64; do
#     for py in 313 314; do
#       pip download --only-binary=:all: --no-deps --dest wheels \
#         --platform $plat --python-version $py --implementation cp --abi cp$py \
#         dbus-fast==<ver> pycryptodome==<ver>
#     done
#   done
#   pip download --only-binary=:all: --no-deps --dest wheels bleak==<ver>
#   for f in wheels/*.whl; do echo "$f"; pip hash "$f"; done
#
# Equivalently: pip-compile --generate-hashes --allow-unsafe requirements.in
# Then verify: python3 -m venv /tmp/v && /tmp/v/bin/pip install \
#   --require-hashes --only-binary=:all: --no-deps -r requirements.txt
